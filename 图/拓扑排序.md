## 拓扑排序代码

```c
// 拓扑排序算法
int topoLogicalSort(MGraph G) {
    int S[MAXNUM];  // 栈，用于存储入度为0的顶点
    int inDegree[MAXNUM];  // 入度数组
    int top = 0;  // 栈顶指针
    int n = G.numVertices;  // 顶点数
    int i = 0, j = 0, in = 0;

    // 1. 计算所有结点的入度，并将入度为0的结点压入栈中
    for (i = 0; i < n; i++) {
        in = 0;
        for (j = 0; j < n; j++) {
            if (G.Edge[j][i] != INF) { // 存在从j到i的边，i的入度增加
                in++;
            }
        }
        // 如果入度为0，压入栈中
        if (in == 0) {
            S[top++] = i;  // 将结点i压入栈
        }
        inDegree[i] = in;  // 记录结点i的入度
    }

    int count = 0;  // 已经输出的结点数
    int printQue[MAXNUM];
    int v;

    // 2. 进行拓扑排序
    while (top != 0) {
        v = S[--top];  // 从栈中弹出一个结点
        printQue[count++]=v ;  // 输出结点数加1

        // 3. 更新与当前结点v相邻的所有结点的入度
        for (i = 0; i < n; i++) {
            if (G.Edge[v][i] != INF) {  // 如果存在边 v -> i
                inDegree[i]--;  // 结点i的入度减少
                if (inDegree[i] == 0) {  // 如果i的入度变为0，压入栈
                    S[top++] = i;  // 将结点i压入栈
                }
            }
        }
    }
    printf("拓扑排序输出顺序：");
    for ( i = 0; i <count; i++)
    {
        printf("%c ", G.VerticesList[printQue[i]]);
    }
    // 4. 如果输出的结点数小于总结点数，说明图中存在环
    if (count != n) {
        printf("\n图中存在环，无法完成拓扑排序。\n");
        return 1;  // 返回错误代码，表示图中有环
    }

    return 0;  // 成功完成拓扑排序
}

```
