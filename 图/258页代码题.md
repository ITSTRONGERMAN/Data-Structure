## 试编写利用DFS实现有向无环图拓扑排序算法

```c
// DFS实现拓扑排序的辅助函数
void DFS(MGraph G, int current, int visited[], int* time, int finishTime[]) {
    visited[current] = 1; // 标记当前顶点为已访问
    printf("%c ", G.VerticesList[current]); // 输出当前顶点
    // 遍历与当前顶点相连的所有邻接点
    for (int i = 0; i < G.numVertices; i++) {
        // 如果没有访问过并且存在从当前顶点到i的边
        if (!visited[i] && G.Edge[current][i] < INF) {
            DFS(G, i, visited, time, finishTime); // 递归访问邻接点
        }
    }
    // 递归结束后标记当前顶点的完成时间
    finishTime[current] = ++(*time);
}

// 拓扑排序主函数
void topoSort_DFS(MGraph G) {
    int visited[MAXNUM] = { 0 };  // 初始化访问标志数组，所有顶点初始时未访问
    int finishTime[MAXNUM] = { 0 }; // 用来记录每个顶点的完成时间
    int i = 0, time = 0;

    // 对每个顶点进行DFS遍历，未访问的顶点才开始DFS
    for (i = 0; i < G.numVertices; i++) {
        if (!visited[i]) {
            DFS(G, i, visited, &time, finishTime); // 从未访问的顶点开始DFS
        }
    }
    printf("\n");

    // 输出所有顶点的完成时间，按完成时间倒序输出即为拓扑排序
    for (i = 0; i < G.numVertices; i++) {
        printf("%d ", finishTime[i]);
    }
    printf("\n");

    // 为了输出拓扑排序，可以根据完成时间进行排序，得到拓扑排序
    // 基于完成时间的倒序排序
    int sortedVertices[MAXNUM];
    for (i = 0; i < G.numVertices; i++) {
        sortedVertices[i] = i;
    }

    // 简单的冒泡排序根据完成时间倒序排列
    for (i = 0; i < G.numVertices - 1; i++) {
        for (int j = i + 1; j < G.numVertices; j++) {
            if (finishTime[sortedVertices[i]] < finishTime[sortedVertices[j]]) {
                // 交换顶点顺序
                int temp = sortedVertices[i];
                sortedVertices[i] = sortedVertices[j];
                sortedVertices[j] = temp;
            }
        }
    }

    // 输出拓扑排序后的结果，按照完成时间的倒序
    printf("Topological Sort: ");
    for (i = 0; i < G.numVertices; i++) {
        printf("%c ", G.VerticesList[sortedVertices[i]]);
    }
    printf("\n");
}
```
