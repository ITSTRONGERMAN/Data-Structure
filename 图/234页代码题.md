## 试设计一个算法，判断一个无向图G是否是一棵树。若是一棵树，则算法返回true，否则返回false

```c
// 遍历计算结点数和边数
void DFS_isTree(AjGraph G, int* visited, int start, int* vnum, int* edgeNum) {
    visited[start] = 1;
    (*vnum)++;
    for (ArcNode* p = G.vertices[start].firstarc; p; p = p->nextarc) {
            (*edgeNum)++;
            if (visited[p->adjvex] == 0) {
                DFS_isTree(G, visited, p->adjvex, vnum, edgeNum);
            }
    }
}
// 是否是一颗树
bool isTree(AjGraph G) {
    int* visited = (int*)malloc(sizeof(int) * G.vexnum);  // 分配内存
    if (visited == NULL) {
        printf("visited数组内存分配失败\n");
        return;
    }
    // 初始化visited数组为0，表示所有顶点未访问
    for (int i = 0; i < G.vexnum; i++) {
        visited[i] = 0;
    }
    int vnum = 0, edgeNum = 0;
    //通过深度优先遍历算法,统计结点数和边数
    DFS_isTree(G, visited, 0, &vnum, &edgeNum);
    printf("%d %d", vnum, edgeNum);
    if (vnum == G.vexnum && edgeNum == 2 * (G.vexnum - 1)) return true;
    else return false;
}
```

## 分别基于深度优先遍历和广度优先遍历算法判别以邻接表方式存储的有向图中是否存在由顶点vi到顶点vj的路径（i!=vj）。注意算法涉及图的基本操作必须在此存储结构中实现。

深度优先遍历：

```c
// 深度优先搜索 两个顶点是否存在路径
void DFS_isLink(AjGraph G, int from, int to, bool* can_reach, int* visited) {
    if (from == to) {
        *can_reach = true;  // 如果找到目标点，设置can_reach为true
        return;
    }
    visited[from] = 1;  // 标记当前顶点已访问
    for (ArcNode* p = G.vertices[from].firstarc; p; p = p->nextarc) {
        if (visited[p->adjvex] == 0 && !(*can_reach)) {  // 如果邻接点未被访问且还没有找到路径
            DFS_isLink(G, p->adjvex, to, can_reach, visited);
        }
    }
}

// 检查两个顶点是否连通
bool isLink_DFS(AjGraph G, int from, int to) {
    int* visited = (int*)malloc(sizeof(int) * G.vexnum);  // 分配内存
    if (visited == NULL) {
        printf("visited数组内存分配失败\n");
        return false;  // 内存分配失败时返回false
    }

    bool can_reach = false;
    // 初始化visited数组为0，表示所有顶点未访问
    for (int i = 0; i < G.vexnum; i++) {
        visited[i] = 0;
    }

    DFS_isLink(G, from, to, &can_reach, visited);

    printf("%d到%d%s路径\n", from, to, can_reach ? "有" : "没有");

    free(visited);  // 释放内存
    return can_reach;
}
```

```c
// 广度优先遍历 两个顶点是否存在路径
bool BFS_isLink(AjGraph G, int from, int to) {
      int* visited = (int*)malloc(sizeof(int) * G.vexnum);  // 分配内存
    if (visited == NULL) {
        printf("visited数组内存分配失败\n");
        return false;  // 内存分配失败时返回false
    }
    // 初始化visited数组为0，表示所有顶点未访问
    for (int i = 0; i < G.vexnum; i++) {
        visited[i] = 0;
    }
    int v;
    int Q[100];
    int front = 0, rear = 0;
    Q[rear++] = from;
    while (front != rear) {
        v = Q[front++];
        if (v == to)return true;
        visited[v] = 1;
        for (ArcNode* p=G.vertices[v].firstarc;p;p=p->nextarc)
        {
            if (p->adjvex == to) return true;

            if (visited[p->adjvex]==0) {
                Q[rear++] = p;
                visited[p->adjvex] = 1;
            }
        }
    }
    return false;
}
```

## 假设图用邻接表表示，设计一个算法，输出从顶点V1到带那顶点V2的所有简单路径

```c
// 打印路径的函数
void printPath(AjGraph G,int path[], int length) {
    for (int i = 0; i <= length; i++) {
        printf("%c ", G.vertices[path[i]].data);
    }
    printf("\n");
}

// 深度优先搜索查找简单路径
void findSimplePath(AjGraph G, int vi, int vj, int path[], int* d, int visited[]) {
    // 将当前顶点加入路径
    path[(*d)] = vi;
    (*d)++;  // 增加路径长度
    visited[vi] = 1; // 标记当前顶点为已访问

    // 如果找到目标顶点，打印路径
    if (vi == vj) {
        printPath(G,path, (*d) - 1);  // 打印从起点到目标顶点的路径（不包括目标点后的部分）
    }else{
     // 遍历邻接点，递归查找路径
    for (ArcNode* p = G.vertices[vi].firstarc; p != NULL; p = p->nextarc) {
        if (visited[p->adjvex] == 0) {  // 如果邻接点没有访问过
            findSimplePath(G, p->adjvex, vj, path, d, visited);
        }
    }   
    }
    // 回溯时，恢复状态
    (*d)--;  // 恢复路径长度
    visited[vi] = 0; // 恢复当前顶点为未访问状态
}
```
