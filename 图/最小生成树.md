## Prim算法

```c
// 邻接矩阵
// Prim算法实现最小生成树
void Prim_MG(MGraph G) {
    int lowCost[MAXNUM];       // 存储到树的最短边权值
    int prevVertex[MAXNUM];    // 存储树中每个顶点的前驱节点
    int inTree[MAXNUM];        // 标记顶点是否已经在树中
    int i, j, minIndex, min;   // min：dang
    // 初始化
    for (i = 0; i < G.numVertices; i++) {
        lowCost[i] = G.Edge[0][i];   // 初始时，将第0个顶点到其他顶点的边权值赋值
        prevVertex[i] = 0;           // 前驱节点初始为0
        inTree[i] = 0;               // 顶点不在树中
    }
    lowCost[0] = 0;  // 第一个顶点的权值设为0，表示它是起始顶点
    inTree[0] = 1;   // 将第一个顶点加入树中

    // 遍历所有顶点，直到所有顶点都加入到树中
    for (i = 1; i < G.numVertices; i++) {
        min = INF;
        minIndex = -1;

        // 找出在树外的顶点中，权值最小的边
        for (j = 0; j < G.numVertices; j++) {
            if (!inTree[j] && lowCost[j] < min) {
                min = lowCost[j];
                minIndex = j;
            }
        }

        // 将minIndex顶点加入最小生成树
        inTree[minIndex] = 1;

        // 更新与minIndex相邻的顶点的lowCost和prevVertex
        for (j = 0; j < G.numVertices; j++) {
            // 如果从minIndex到j的边权值更小，更新lowCost和prevVertex
            if (!inTree[j] && G.Edge[minIndex][j] < lowCost[j]) {
                lowCost[j] = G.Edge[minIndex][j];
                prevVertex[j] = minIndex;
            }
        }
    }

    // 输出最小生成树
    printf("最小生成树的边:\n");
    for (i = 1; i < G.numVertices; i++) {
        printf("顶点 %c - 顶点 %c，权值 = %d\n", 
               G.VerticesList[prevVertex[i]], 
               G.VerticesList[i], 
               G.Edge[prevVertex[i]][i]
              );
    }
}
```

## Kruskal算法

```c
// 用于Kruskal算法的边结构体
typedef struct {
    int begin;
    int end;
    int weight;
}Edge;

// 并查集初始化函数
void initParent(int parent[], int n) {
    for (int i = 0; i < n; i++) {
        parent[i] = i;  // 每个顶点的父节点初始为自己
    }
}

// 查找操作，带路径压缩
int find(int parent[], int x) {
    if (parent[x] != x) {
        parent[x] = find(parent, parent[x]);  // 递归查找并进行路径压缩
    }
    return parent[x];
}

// 合并操作
void unionSet(int parent[], int root1, int root2) {
    parent[root1] = root2;  // 将root1的根连接到root2上
}

// Kruskal算法实现
void Kruskal(MGraph G, Edge* E) {
    int i, j, k, parent[MAXNUM], temp;
    int edgeCount = 0;  // 记录最小生成树的边数

    // 初始化并查集
    initParent(parent, G.numVertices);

    // 从图的邻接矩阵中提取所有边信息
    k = 0;  // 边表的索引
    for (i = 0; i < G.numVertices; i++) {
        for (j = i + 1; j < G.numVertices; j++) {  // 只遍历上三角部分（无向图）
            if (G.Edge[i][j] != INF) {
                E[k].begin = i;
                E[k].end = j;
                E[k].weight = G.Edge[i][j];
                k++;  // 增加边的数量
            }
        }
    }

    // 边表排序（冒泡排序）
    for (i = 0; i < G.numEdges - 1; i++) {
        for (j = i + 1; j < G.numEdges; j++) {
            if (E[i].weight > E[j].weight) {
                // 交换边
                Edge tempEdge = E[i];
                E[i] = E[j];
                E[j] = tempEdge;
            }
        }
    }

    // 选取最小的边，构造最小生成树
    for (i = 0; i < G.numEdges; i++) {
        int root1 = find(parent, E[i].begin);  // 查找第一个端点的根
        int root2 = find(parent, E[i].end);    // 查找第二个端点的根

        if (root1 != root2) {  // 如果这两个端点不在同一个连通块中
            printf(
                "Edge: (%c, %c) Weight: %d\n",
                G.VerticesList[E[i].begin], 
                G.VerticesList[E[i].end], 
                E[i].weight
            );  // 选择这条边
            unionSet(parent, root1, root2);  // 合并这两个顶点所在的集合
            edgeCount++;  // 增加最小生成树中的边数

            if (edgeCount == G.numVertices - 1) {  // 如果已经选取了足够的边，结束
                break;
            }
        }
    }
}
```
