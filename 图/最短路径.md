## Dijkstra算法

```c
// Dijkstra 算法实现
void Dijkstra(MGraph G, int start) {
    int i, j;
    int distance[MAXNUM];   // 存储从起始点到每个节点的最短距离
    int path[MAXNUM];       // 存储最短路径上的前驱节点
    int final[MAXNUM];      // final[i]为1表示节点i已加入最短路径集，0表示未加入

    // 初始化
    for (i = 0; i < G.numVertices; i++) {
        distance[i] = G.Edge[start][i];   // 起始节点到各点的初始距离为邻接矩阵中的值
        if (G.Edge[start][i] < INF) {
            path[i] = start;  // 初始路径为从start节点到各节点
        }
        else {
            path[i] = INF;     // 没有直接路径时，设置为-1
        }
        final[i] = 0;      // 初始化所有节点都没有加入最短路径集
    }

    distance[start] = 0;    // 起始点到自己的距离为0
    final[start] = 1;    // 起始点加入最短路径集
    int min, minIndex;
    // 从起始节点开始，逐步更新最短路径
    for (i = 0; i < G.numVertices - 1; i++) {
         min = INF;     // 当前未加入最短路径集的节点中，最短的距离
         minIndex = -1; // 存储最短路径的节点

        // 遍历所有节点，找出距离起始点最近的节点
        for (j = 0; j < G.numVertices; j++) {
            if (!final[j] && distance[j] < min) {
                min = distance[j];
                minIndex = j;
            }
        }

        // 将该节点加入最短路径集
        final[minIndex] = 1;

        // 更新该节点邻接点的最短距离
        for (j = 0; j < G.numVertices; j++) {
            // 如果节点j没有加入最短路径集，并且从minIndex到j的距离更短
            if (!final[j] && G.Edge[minIndex][j] < INF) {
                // 更新最短路径
                if (distance[minIndex] + G.Edge[minIndex][j] < distance[j]) {
                    distance[j] = distance[minIndex] + G.Edge[minIndex][j];
                    path[j] = minIndex;  // 更新路径的前驱节点
                }
            }
        }
    }

    // 输出最短路径
    for (i = 0; i < G.numVertices; i++) {
        if (i != start) {
            printf(
                "顶点%c 到 顶点%c, 最短路径长度：%d. Path: ", 
                G.VerticesList[start], 
                G.VerticesList[i], 
                distance[i]
            );
            int p = i;
            while (p != start) {
                printf("%c <- ", G.VerticesList[p]);
                p = path[p];
            }
            printf("%c\n", G.VerticesList[start]);
        }
    }
}
```

## Floyd算法

```c
// 打印最短路径矩阵
void printSolution(int dist[MAXNUM][MAXNUM], int n) {
    printf("最短路径矩阵：\n");
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (dist[i][j] == INF) {
                printf("INF ");  // 无法到达的节点，显示INF
            }
            else {
                printf("%d ", dist[i][j]);  // 打印最短路径长度
            }
        }
        printf("\n");
    }
}

// Floyd算法实现
void Floyd(MGraph G) {
    int dist[MAXNUM][MAXNUM];
    int n = G.numVertices; // 获取顶点数量
    // 初始化dist数组
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) {
                dist[i][j] = 0;  // 从一个顶点到自己，距离为0
            }
            else if (G.Edge[i][j] == INF) {
                dist[i][j] = INF;  // 无边，设为无穷大
            }
            else {
                dist[i][j] = G.Edge[i][j];  // 有边，使用该边的权值
            }
        }
    }

    // Floyd核心算法
    // 通过逐个中间节点k来更新所有i到j的路径
    for (int k = 0; k < n; k++) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (dist[i][j] > dist[i][k] + dist[k][j]) {
                    dist[i][j] = dist[i][k] + dist[k][j];  // 更新最短路径
                }
            }
        }
    }
    printSolution(dist, n);
}
```
