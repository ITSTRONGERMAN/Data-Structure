## 邻接矩阵结构相关操作

```c
typedef struct {
	int numVertices, numEdges; // 定点数和边数
	char VerticesList[MAXVERTEXNUM]; // 顶点表
	int Edge[MAXVERTEXNUM][MAXVERTEXNUM]; // 边表
}MGraph; // 矩阵
// 初始化图
void InitGraph(MGraph* G) {
    G->numVertices = 0;
    G->numEdges = 0;

    // 初始化边矩阵，将所有边设置为INFINITY（表示无边）
    for (int i = 0; i < MAXVERTEXNUM; i++) {
        for (int j = 0; j < MAXVERTEXNUM; j++) {
              G->Edge[i][j] = 0; // 自己到自己的距离为0
        }
    }
}
// 插入顶点
void InsertVertex(MGraph* G, char vertex) {
    if (G->numVertices < MAXVERTEXNUM) {
        G->VerticesList[G->numVertices] = vertex;
        G->numVertices++;
    }
    else {
        printf("图的顶点已满，无法插入新顶点\n");
    }
}

// 插入边(style为1是有向图，0为无向图)
void InsertEdge(MGraph* G, char startVertex, char endVertex, int weight,int style) {
    int startIndex = -1, endIndex = -1;

    // 找到起始顶点的索引
    for (int i = 0; i < G->numVertices; i++) {
        if (G->VerticesList[i] == startVertex) {
            startIndex = i;
        }
        if (G->VerticesList[i] == endVertex) {
            endIndex = i;
        }
    }

    // 如果找到了起始顶点和终止顶点，插入边
    if (startIndex != -1 && endIndex != -1) {
        if (style == 1) {
            G->Edge[startIndex][endIndex] = weight;
        }
        if (style==0) {
            G->Edge[startIndex][endIndex] = weight;
            G->Edge[endIndex][startIndex] = weight;
        }
        G->numEdges++;
    }
    else {
        printf("顶点未找到，无法插入边\n");
    }
}
// 打印图（邻接矩阵）
void PrintGraph(MGraph* G) {
    printf("图的顶点：\n");
    for (int i = 0; i < G->numVertices; i++) {
        printf("%c ", G->VerticesList[i]);
    }
    printf("\n");

    printf("图的邻接矩阵：\n");
    for (int i = 0; i < G->numVertices; i++) {
        for (int j = 0; j < G->numVertices; j++) {
            if (G->Edge[i][j] == MAXINT)
                printf("INF ");
            else
                printf("%d ", G->Edge[i][j]);
        }
        printf("\n");
    }
}
// 广度优先遍历（邻接矩阵）
void BFSMGraph(MGraph G, int start) {
    printf("邻接矩阵的广度优先遍历算法：");

    // 创建访问标记数组
    int* visited = (int*)malloc(sizeof(int) * G.numVertices);
    if (visited == NULL) {
        printf("visited数组内存分配失败\n");
        return;
    }

    // 初始化visited数组，所有顶点未访问
    for (int i = 0; i < G.numVertices; i++) {
        visited[i] = 0;
    }

    // 队列的初始化
    int Q[MAXVERTEXNUM];  // 辅助队列
    int front = 0, rear = 0;  // 队头队尾指针

    // 输出起始节点
    printf("%c ", G.VerticesList[start]);
    visited[start] = 1;
    Q[rear++] = start;  // 将起始节点加入队列

    int v; // 当前结点序号
    while (front != rear) {
        v = Q[front++];  // 队头出队
        // 遍历当前节点的所有邻接节点
        for (int w = 0; w < G.numVertices; w++) {
            // 如果w未被访问且存在边 (v, w)
            if (visited[w] == 0 && G.Edge[v][w] == 1) {
                printf("%c ", G.VerticesList[w]);
                visited[w] = 1;
                Q[rear++] = w;  // 将未访问的邻接点加入队列
            }
        }
    }

    free(visited);  // 释放内存
}
// 深度优先遍历的递归实现（邻接矩阵）
void DFS_M(MGraph G, int start, int* visited) {
    printf("%c ", G.VerticesList[start]);  // 输出当前顶点
    visited[start] = 1;  // 标记当前顶点为已访问

    // 遍历当前顶点的所有邻接点
    for (int i = 0; i < G.numVertices; i++) {
        if (visited[i] == 0 && G.Edge[start][i] == 1) {  // 如果没有访问过且有边
            DFS_M(G, i, visited);  // 递归访问该顶点
        }
    }
}

// 初始化并调用深度优先遍历
void DFS_M_Graph(MGraph G, int start) {
    printf("邻接矩阵的深度优先遍历：");
    int* visited = (int*)malloc(sizeof(int) * G.numVertices);  // 分配内存

    if (visited == NULL) {
        printf("visited数组内存分配失败\n");
        return;
    }

    // 初始化visited数组为0，表示所有顶点未访问
    for (int i = 0; i < G.numVertices; i++) {
        visited[i] = 0;
    }

    // 调用DFS进行深度优先遍历
    DFS_M(G, start, visited);

    // 完成遍历后，释放内存
    free(visited);
}
```

## 邻接表相关操作

```c
// 邻接表
typedef struct ArcNode { // 边表结点
    int adjvex; // 该弧所指向的顶点的位置
    struct ArcNode* nextarc; // 指向下一条弧的指针
}ArcNode;
typedef struct VNode { // 顶点表结点
    vertexType data; // 顶点信息
    ArcNode* firstarc; // 指向第一条依附该节点的弧的指针
}VNode,AdjList[MAXVERTEXNUM];
typedef struct {
    AdjList vertices; //邻接表
    int vexnum, arcnum; // 顶点数和边数
}AjGraph;
// 初始化图
void InitGraph(AjGraph* G, int vertexCount) {
    G->vexnum = 0;
    G->arcnum = 0;
    for (int i = 0; i < MAXVERTEXNUM; i++) {
        G->vertices[i].firstarc = NULL;  // 每个顶点的边表初始化为空
        G->vertices[i].data = '\0';      // 顶点的数据初始化为空字符（可以根据需求修改）
    }
}
// 添加邻接表结点
void AddAjGraphVertex(AjGraph* G, char vertexData) {
    if (G->vexnum >= MAXVERTEXNUM) {
        printf("图的顶点数已经达到最大值，无法添加新顶点！\n");
        return;
    }

    // 初始化新顶点
    int index = G->vexnum;
    G->vertices[index].data = vertexData;  // 设置顶点数据
    G->vertices[index].firstarc = NULL;    // 初始化邻接表为空链表

    // 更新顶点数量
    G->vexnum++;
}
// 添加邻接表边（无向图）
void AddAjGraphEdge(AjGraph* G, int start, int end) {
    // 创建一条从start到end的边
    ArcNode* newEdge = (ArcNode*)malloc(sizeof(ArcNode));
    newEdge->adjvex = end;
    newEdge->nextarc = G->vertices[start].firstarc;
    G->vertices[start].firstarc = newEdge;  // 将新边插入start的邻接表

    // 创建一条从end到start的边（无向图）
    newEdge = (ArcNode*)malloc(sizeof(ArcNode));
    newEdge->adjvex = start;
    newEdge->nextarc = G->vertices[end].firstarc;
    G->vertices[end].firstarc = newEdge;  // 将新边插入end的邻接表

    // 更新边的数量
    G->arcnum++;
}
// 打印图的邻接表
void printAjGraph(AjGraph G) {
    for (int i = 0; i < G.vexnum; i++) {
        printf("Vertex %d (%c):", i, G.vertices[i].data);
        ArcNode* p = G.vertices[i].firstarc;
        while (p != NULL) {
            printf(" -> %d", p->adjvex);
            p = p->nextarc;
        }
        printf("\n");
    }
}
// 广度优先遍历算法(邻接表)
void BFSAjGraph(AjGraph G, int start) {
    printf("邻接表的广度优先遍历算法：");
    int* visited = (int*)malloc(sizeof(int) * G.vexnum);  // 创建访问标记数组
    if (visited == NULL) {
        printf("visited数组内存分配失败\n");
        return;
    }
    // 初始化visited数组，所有顶点未访问
    for (int i = 0; i < G.vexnum; i++) {
        visited[i] = 0;
    }
    int Q[MAXVERTEXNUM]; // 辅助队列
    int front = 0, rear = 0; // 队头队尾指针
    Q[rear++] = start;   // 将起始节点加入队列
    visited[start] = 1;   // 标记起始节点为已访问
    printf("%c ", G.vertices[start].data); // 输出起始顶点
    int v; // 当前顶点
    int w; // 邻接顶点
    ArcNode* p; // 边的指针
    while (front != rear) {
        v = Q[front++];  // 队头出队
        for (p = G.vertices[v].firstarc; p; p = p->nextarc) {
            w = p->adjvex;
            if (visited[w] == 0) { // 如果未访问
                printf("%c ", G.vertices[w].data);  // 输出当前顶点
                visited[w] = 1;     // 标记已访问
                Q[rear++] = w;      // 加入队列
            }
        }
    }

    free(visited);  // 释放内存
}
// 深度优先遍历（邻接表）
// DFS递归函数
void DFS(AjGraph G, int start, int* visited) {
    // 访问当前顶点
    printf("%c ", G.vertices[start].data);
    visited[start] = 1; // 标记当前结点

    // 遍历当前顶点的所有邻接点
    for (ArcNode* p = G.vertices[start].firstarc; p; p = p->nextarc) {
        if (visited[p->adjvex] == 0) {  // 如果邻接点未被访问
            DFS(G, p->adjvex, visited);  // 递归访问邻接点
        }
    }
}
// 深度优先遍历
void DFS_AJ_Graph(AjGraph G, int start) {
    printf("深度优先遍历：");

    if (visited == NULL) {
        printf("visited数组内存分配失败\n");
        return;
    }

    // 初始化visited数组，所有顶点未访问
    for (int i = 0; i < G.vexnum; i++) {
        visited[i] = 0;
    }

    // 调用DFS进行深度优先遍历
    DFS(G, start, visited);

    // 完成遍历后，释放visited数组内存

}
```
