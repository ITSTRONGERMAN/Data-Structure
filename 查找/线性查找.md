## 顺序查找

```c
typedef int ElementType;
typedef struct {
	ElementType data[MAX];
	int length;
} LinearSearchTable; // 线性搜索表
void insert_LST(LinearSearchTable* T,ElementType data) {
	T->data[T->length++] = data;
}
void init_LST(LinearSearchTable* T) {
	T->data[0] = 0;
	T->length = 1;
}
int search_LST(LinearSearchTable T,ElementType target) {
	T.data[0] = target;
	int i = 0;
	for (i = T.length; T.data[i] != target && i > 0; i--);
	return i;
}
```

## 折半查找（二分查找）

```c
int binarySearch(LinearSearchTable T,ElementType target) {
	int low = 1, high = T.length - 1, mid;
	while (low <= high) {
		mid = (low + high) / 2;
		if (T.data[mid] > target)  high = mid - 1;
		else if (T.data[mid] < target) low = mid + 1;
		else return mid;
	}
	return -1;
}

```
