```c
int heightOfBinaryTree(BinaryTree T){
    if (T == NULL) return 0;
    else {
        int lh = heightOfBinaryTree(T->lchild);
        int rh = heightOfBinaryTree(T->rchild);
        return (lh > rh ? lh : rh) + 1;
    }

}
```

```c
BinaryTree deleteLeaveNode(BinaryTree T) {
    if (T == NULL) return NULL;
    if (T->lchild == NULL && T->rchild == NULL) {
        free(T);
        return NULL;
    }
    T->lchild = deleteLeaveNode(T->lchild);
    T->rchild = deleteLeaveNode(T->rchild);
    return T;
}
```

## 计算指定结点*p所在的层次

```c
int getNodeLayer(BinaryTree T, int element) {
    BinaryTreeNode* Q[100];
    int front = -1, rear = -1;
    BinaryTreeNode* p;
    int layerLast = 0; // 每一层的最后一个元素
    int layer = 1;
    Q[++rear] = T;
    while (front!=rear)
    {
        p = Q[++front];
        if (p->data == element) return layer;
        if (p->lchild != NULL) Q[++rear] = p->lchild;
        if (p->rchild != NULL) Q[++rear] = p->rchild;
        if (layerLast == front) {
            layer++;
            layerLast = rear;
        }
    }
    return -1;
}
```

## 计算二叉树中各结点中的最大元素的值

```c
int getMaxValue(BinaryTree T) {
    if (T == NULL) {
        return INT_MIN;
    }
    int leftMax = getMaxValue(T->lchild);
    int rightMax = getMaxValue(T->rchild);
    int maxValue = T->data > leftMax ? T->data : leftMax;
    maxValue = maxValue > rightMax ? maxValue : rightMax;
    return maxValue;
}
```
